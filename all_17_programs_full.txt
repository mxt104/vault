========================================
<!DOCTYPE html>
<html>
<head>
<title>Grade</title>
<style>
body{text-align:center;font-family:Arial}
input,button{margin:5px;padding:8px}
</style>
</head>
<body>
<h2>Grade Calculator</h2>
<input id="m1" placeholder="Marks 1">
<input id="m2" placeholder="Marks 2">
<input id="m3" placeholder="Marks 3"><br>
<button onclick="calc()">Calculate</button>
<p id="res"></p>
<script>
function calc(){
let a=+m1.value,b=+m2.value,c=+m3.value;
let t=a+b+c,p=t/3,g;
if(p>=90)g="A";
else if(p>=75)g="B";
else if(p>=60)g="C";
else if(p>=40)g="D";
else g="F";
res.innerText="Total:"+t+" %:"+p.toFixed(2)+" Grade:"+g;
}
</script>
</body>
</html>

cToF(c){return c*9/5+32;}

}
let t=new Temp();
c.oninput=function(){
if(isNaN(c.value)){msg.innerText="Invalid";return;}
f.value=t.cToF(c.value);
};
f.oninput=function(){
if(isNaN(f.value)){msg.innerText="Invalid";return;}
c.value=t.fToC(f.value);
};
</script>

<script>
class Product{
constructor(n,p,q){this.n=n;this.p=p;this.q=q;}
}
let arr=[
new Product("A",100,2),
new Product("B",200,1),
new Product("C",50,5),
new Product("D",80,3),
new Product("E",120,2)
];
arr.forEach(x=>{
t.innerHTML+=`<tr><td>${x.n}</td><td>${x.p}</td><td>${x.q}</td></tr>`;

function total(){
let s=0;
arr.forEach(x=>s+=x.p*x.q);
if(s>500)s*=0.9;
res.innerText="Total:"+s;
}
