(vi) Factorial (For & While Loop)
▶ 
#!/bin/bash
echo "Enter number:"
read n
# Using For loop
fact_f=1
for ((i=1; i<=n; i++)); do
    fact_f=$((fact_f * i))
done
echo "Factorial (For): $fact_f"

# Using While loop
fact_w=1
temp=$n
while [ $temp -gt 1 ]; do
    fact_w=$((fact_w * temp))
    temp=$((temp - 1))
done
echo "Factorial (While): $fact_w"
(vii) Armstrong Number
▶ 
#!/bin/bash
echo "Enter number:"
read n
sum=0
temp=$n
while [ $temp -gt 0 ]; do
    digit=$((temp % 10))
    sum=$((sum + digit**3))
    temp=$((temp / 10))
done
if [ $sum -eq $n ]; then
    echo "Armstrong number"
else
    echo "Not Armstrong"
fi

▶

Practical 8: Advanced Shell Scripts

(i) Search Element in List
▶ 
#!/bin/bash
echo "Enter elements (space separated):"
read -a list
echo "Enter element to search:"
read search
found=0
for i in "${list[@]}"; do
    if [ "$i" == "$search" ]; then
        found=1
        break
    fi
done
if [ $found -eq 1 ]; then echo "Present"; else echo "Not Present"; fi

(ii) Perl: Prime Check
▶ 
#!/usr/bin/perl
print "Enter number: ";
$n = <STDIN>;
$flag = 0;
for($i=2; $i<=$n/2; $i++) {
    if($n % $i == 0) { $flag = 1; last; }
}
if($flag == 0 && $n > 1) { print "Prime\n"; } else { print "Not Prime\n"; }

▶

BEGIN {
    print "Enter start and end range:"
    getline start < "-"
    getline end < "-"
    for(i=start; i<=end; i++) {
        if(i % 2 == 0) print i
    }
}

    print "Enter number of terms:"

        print a;
        c=a+b; a=b; b=c;
    }
}
