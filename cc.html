6Check Prime Number ▶ 
#!/bin/bash
echo "enter number"
read n
if [ $n -le 1 ]
then
    echo "$n is not a prime number"
    exit
fi

i=2
while [ $i -lt $n ]
do
    if [ $((n%i)) -eq 0 ]
    then
        echo "$n is not a prime number"
        exit
    fi
    i=$((i+1))
done

echo "$n is a prime number"

9Armstrong Number ▶ 
#!/bin/bash
echo "enter a number"
read n

original=$n
sum=0

while [ $n -gt 0 ]
do
    digit=$((n%10))
    sum=$((sum + digit*digit*digit))
    n=$((n/10))
done

if [ $sum -eq $original ]
then
    echo "armstrong number"
else
    echo "not armstrong"
fi

7Count Files & Directories ▶

read d
